# =============================================================================
# TARODAN INFRASTRUCTURE ENVIRONMENT VARIABLES

# GHCR image source (set by deploy job / server .env)
REGISTRY=ghcr.io/sigmoida
IMAGE_TAG=latest

# Domain Configuration
DOMAIN=tarodan.com
ACME_EMAIL=admin@tarodan.com

# Traefik Dashboard Auth (generate with: htpasswd -nb admin password)
TRAEFIK_AUTH=admin:$$apr1$$xyz...

# Adminer (canlı DB görüntüleyici) basic-auth — https://db.$DOMAIN için.
# Üret: htpasswd -nb <kullanici> <sifre>  (çıktıdaki tek $ işaretlerini $$ yap)
ADMINER_AUTH=admin:$$apr1$$xyz...

# PostgreSQL Database
DB_USER=tarodan
DB_PASSWORD=your_secure_database_password_here
DB_NAME=tarodan

# Redis
REDIS_PASSWORD=your_secure_redis_password_here
# Rol ayrımı: API'yi cron tüketiminden ayırmak için `web` yapın (işler worker'a
# kalır). Varsayılan `all` tek-process davranışını korur.
API_PROCESS_ROLE=all

# JWT Secret (generate with: openssl rand -base64 64)
JWT_SECRET=your_jwt_secret_key_here

# AWS S3 Object Storage
# GitHub Secrets'tan veya .env dosyasından alınır
AWS_ACCESS_KEY_ID=your_aws_access_key_id_here
AWS_SECRET_ACCESS_KEY=your_aws_secret_access_key_here
AWS_REGION=eu-west-1
S3_BUCKET=amzn-tarodan
S3_ENV_PREFIX=prod

# Grafana Dashboard
GRAFANA_USER=admin
GRAFANA_PASSWORD=your_secure_grafana_password_here

# External Service API Keys
# iyzico Payment Gateway
IYZICO_API_KEY=your_iyzico_api_key
IYZICO_SECRET_KEY=your_iyzico_secret_key
IYZICO_BASE_URL=https://api.iyzipay.com

# Email (SMTP — tarodan.com.tr mail hosting)
# Consumed by docker-compose.prod.yml for both the api and worker services.
# Without SMTP_HOST the API boots but silently logs mail instead of sending it.
SMTP_HOST=mail.akilliticaret.com
SMTP_PORT=587
SMTP_USER=info@tarodan.com.tr
SMTP_PASS=your_mailbox_password_here
SMTP_SECURE=false
# All outbound mail is sent as info@; destek@ is only ever a recipient.
MAIL_FROM=Tarodan <info@tarodan.com.tr>
SUPPORT_EMAIL=destek@tarodan.com.tr
SUPPORT_NOTIFICATION_EMAIL=destek@tarodan.com.tr

# Sentry Error Tracking
SENTRY_DSN=https://your_key@sentry.io/your_project
SENTRY_ORG=tarodan
SENTRY_PROJECT=api
LOG_LEVEL=info

# Frontend Sentry
NEXT_PUBLIC_SENTRY_DSN=https://your_key@sentry.io/your_project
